/** * MY LISAA Secure EasyPost Shipping Rates Worker * Environment Variable Required in Cloudflare Worker Settings: * EASYPOST_API_KEY = "EZTK..." or "EZAK..." */ export default { async fetch(request, env, ctx) { // Handle CORS Preflight Options Request if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" } }); } if (request.method !== "POST") { return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { const body = await request.json(); const { to_address, from_address, parcel } = body; if (!to_address || !to_address.country || !to_address.zip) { return new Response(JSON.stringify({ error: "Invalid destination address details." }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // Verify API Key exists in Environment Variables if (!env.EASYPOST_API_KEY) { return new Response(JSON.stringify({ error: "EasyPost API Key missing in server configuration." }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // Construct EasyPost API Payload const easyPostPayload = { shipment: { to_address: { street1: to_address.street1 || "Standard Delivery", city: to_address.city, state: to_address.state, zip: to_address.zip, country: to_address.country }, from_address: { street1: from_address.STREET, city: from_address.CITY, state: from_address.PROVINCE, zip: from_address.POSTAL_CODE, country: from_address.COUNTRY }, parcel: { weight: parseFloat(parcel.weight), length: parseFloat(parcel.length), width: parseFloat(parcel.width), height: parseFloat(parcel.height) } } }; // Secure Authorization header using base64 encoding of API key const authHeader = "Basic " + btoa(env.EASYPOST_API_KEY + ":"); const easyPostResponse = await fetch("https://api.easypost.com/v2/shipments", { method: "POST", headers: { "Authorization": authHeader, "Content-Type": "application/json" }, body: JSON.stringify(easyPostPayload) }); const responseData = await easyPostResponse.json(); if (!easyPostResponse.ok) { return new Response(JSON.stringify({ error: responseData.error?.message || "Failed to fetch rates from EasyPost" }), { status: easyPostResponse.status, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // Filter and clean rates to return to browser safely const cleanRates = (responseData.rates || []).map(r => ({ id: r.id, carrier: r.carrier, service: r.service, rate: r.rate, currency: r.currency, est_delivery_days: r.est_delivery_days })); return new Response(JSON.stringify({ rates: cleanRates }), { status: 200, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (err) { return new Response(JSON.stringify({ error: "Server error calculating shipping: " + err.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } };
